\subsection{Domaine Block World classique}\label{domain_bw}
\begin{lstlisting}[language=Lisp,basicstyle=\ttfamily\footnotesize]
(define (domain bw)
  (:requirements :strips)
  (:constants red green blue)
  (:predicates
      (on ?x ?y)      ; object ?x is on ?object ?y
      (on-table ?x)   ; ?x is directly on the table
      (clear ?x)      ; ?x has nothing on it
      (arm-empty)     ; robot isn't holding anything
      (holding ?x))   ; robot is holding ?x

  ;; les quatres actions classique de manipulation d'objet
  (:action pick-up
     :parameters (?ob1)
     :precondition (and (clear ?ob1) (on-table ?ob1) (arm-empty))
     :effect
     (and (not (on-table ?ob1))
  	     (not (clear ?ob1))
	     (not (arm-empty))
	     (holding ?ob1)))

  (:action put-down
     :parameters (?ob)
     :precondition (holding ?ob)
     :effect
     (and (not (holding ?ob))
	   (clear ?ob)
	   (arm-empty)
	   (on-table ?ob)))

  (:action stack
     :parameters (?sob ?sunderob)
     :precondition (and (holding ?sob) (clear ?sunderob))
     :effect
     (and (not (holding ?sob))
	   (not (clear ?sunderob))
	   (clear ?sob)
	   (arm-empty)
	   (on ?sob ?sunderob)))

  (:action unstack
     :parameters (?sob ?sunderob)
     :precondition (and (on ?sob ?sunderob) (clear ?sob) (arm-empty))
     :effect
     (and (holding ?sob)
	   (clear ?sunderob)
	   (not (clear ?sob))
	   (not (arm-empty))
	   (not (on ?sob ?sunderob))))
)
    \end{lstlisting}

(define (domain poison)

  (:requirements :strips :negative-preconditions :disjunctive-preconditions)

  (:predicates
    (ok ?v)
    (emp ?v)
    (anti ?v)
    (vide ?v)
    (mort ?v)
    (vivant ?v)
  )
  (:action boire_vo
    :parameters (?v)
    :precondition    (and (ok ?v) (not (emp ?v)) 
                      (not (anti ?v)) (not (vide ?v)) 
                      (not (mort ?v)) (not (vivant ?v)))
    :effect (and (not (ok ?v))
                 (vivant ?v))
  )

  (:action emp
    :parameters (?v)
    :precondition   (and (ok ?v) (not (emp ?v)) 
                         (not (anti ?v)) (not (vide ?v))
                         (not (mort ?v)) (not (vivant ?v)))
    :effect (and
              (not (ok ?v))
              (emp ?v))
  )

  (:action vider_vo
    :parameters (?v)
    :precondition   (and (ok ?v) (not (emp ?v)) 
                         (not (anti ?v)) (not (vide ?v))
                         (not (mort ?v)) (not (vivant ?v)))
    :effect (and
              (not (ok ?v))
              (vide ?v))
  )

  (:action anti
    :parameters (?v)
    :precondition   (and (not (ok ?v)) (emp ?v) 
                         (not (anti ?v)) (not (vide ?v))

              (not (emp ?v))
              (anti ?v))
